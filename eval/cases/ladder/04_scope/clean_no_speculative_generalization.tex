\begin{theorem}\label{thm:scope-no-speculation}
For every real number $x$,
\[
  x^2+2x+2\ge 1.
\]
\end{theorem}
\begin{proof}
Completing the square gives
\[
  x^2+2x+2=(x+1)^2+1\ge 1.
\]
\end{proof}
