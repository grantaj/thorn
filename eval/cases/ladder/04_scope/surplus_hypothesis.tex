\begin{theorem}\label{thm:scope-surplus-hypothesis}
If $x\in\mathbb R$ and $x\ge 0$, then $(x-3)^2\ge 0$.
\end{theorem}
\begin{proof}
For every real number $x$, the square $(x-3)^2$ is nonnegative.
\end{proof}
