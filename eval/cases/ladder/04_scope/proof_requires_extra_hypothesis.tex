\begin{theorem}\label{thm:scope-extra-assumption}
For every real number $x$, if $x^2=1$, then $x=1$.
\end{theorem}
\begin{proof}
Suppose additionally that $x>0$. Taking the positive square root of
$x^2=1$ gives $x=1$, as claimed.
\end{proof}
