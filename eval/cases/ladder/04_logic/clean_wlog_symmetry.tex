\begin{theorem}\label{thm:clean-wlog}
For all real numbers $x,y$,
\[
  \max\{x,y\}\ge \frac{x+y}{2}.
\]
\end{theorem}
\begin{proof}
Without loss of generality assume $x\ge y$. The statement is symmetric in
$x$ and $y$. Then $\max\{x,y\}=x$ and
$2x\ge x+y$, which proves the claim.
\end{proof}
