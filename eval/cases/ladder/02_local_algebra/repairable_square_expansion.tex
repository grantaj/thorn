\begin{theorem}\label{thm:repairable-algebra}
For every real number $x$, one has $(x+1)^2\ge 0$.
\end{theorem}
\begin{proof}
Expanding gives
\[
(x+1)^2=x^2+1.
\]
Since $x^2+1\ge 0$, the claim follows.
\end{proof}
