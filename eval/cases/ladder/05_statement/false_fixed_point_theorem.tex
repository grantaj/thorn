\title{A Fixed Point Claim}
\author{Synthetic Thorn Test}
\maketitle

\begin{theorem}\label{thm:false-as-stated}
Every continuous function $f:[0,1]\to\mathbb{R}$ has a fixed point in $[0,1]$.
\end{theorem}
\begin{proof}
The interval $[0,1]$ is compact and connected. Therefore continuity prevents the graph of $f$ from avoiding the diagonal, so $f(x)=x$ for some $x\in[0,1]$.
\end{proof}
