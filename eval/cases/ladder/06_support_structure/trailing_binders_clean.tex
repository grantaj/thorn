\begin{theorem}\label{thm:trailing-binders}
Two pointwise inequalities hold.
\end{theorem}
\begin{proof}
\[
  m \le f(x) \le M
\]
for every $x\in X$.
\[
  0 \le g(x)
\]
for every $x\in X$.
\end{proof}
