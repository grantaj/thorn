\begin{theorem}\label{thm:missing-hypothesis}
For all real numbers $a,x,y$, if $ax=ay$, then $x=y$.
\end{theorem}
\begin{proof}
From $ax=ay$, cancel $a$ from both sides to obtain $x=y$.
\end{proof}
