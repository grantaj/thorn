\begin{theorem}\label{thm:clean-nonzero}
For all real numbers $a,x,y$ with $a\ne 0$, if $ax=ay$, then $x=y$.
\end{theorem}
\begin{proof}
Subtracting gives $a(x-y)=0$. Since $a\ne0$, it follows that $x-y=0$, hence $x=y$.
\end{proof}
