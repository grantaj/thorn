\begin{theorem}\label{thm:integer-cancellation-clean}
For integers $m,n\in\mathbb Z$, if $2m=2n$, then $m=n$.
\end{theorem}
\begin{proof}
Cancel the common factor $2$ to obtain $m=n$.
\end{proof}
