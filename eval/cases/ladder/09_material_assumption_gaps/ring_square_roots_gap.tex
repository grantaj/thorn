\begin{theorem}\label{thm:ring-square-roots-gap}
Let $R$ be an arbitrary ring with identity. For $x\in R$, if $x^2=1$, then $x=1$ or $x=-1$.
\end{theorem}
\begin{proof}
Factoring gives $(x-1)(x+1)=0$. Therefore one factor is zero, and hence $x=1$ or $x=-1$.
\end{proof}
