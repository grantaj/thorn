\begin{theorem}\label{thm:euclidean-pythagoras-clean}
Let $ABC$ be a right triangle in the Euclidean plane, with leg lengths $a,b$ and hypotenuse length $c$. Then
\[
c^2=a^2+b^2.
\]
\end{theorem}
\begin{proof}
By the Pythagorean theorem, the square on the hypotenuse has area equal to the sum of the squares on the legs. Hence $c^2=a^2+b^2$.
\end{proof}
