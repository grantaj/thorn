\begin{theorem}\label{thm:monotone}
Let $(u_j)$ be a decreasing sequence of real numbers that is bounded below. Then
\[
  u_j\to \inf_{k\ge 1} u_k.
\]
\end{theorem}
\begin{proof}
Set $u=\inf_{k\ge 1}u_k$. Given $\varepsilon>0$, the definition of the infimum
gives an index $N$ with $u_N<u+\varepsilon$. Since the sequence is decreasing,
for every $j\ge N$ we have
\[
  u\le u_j\le u_N<u+\varepsilon.
\]
Hence $|u_j-u|<\varepsilon$ for every $j\ge N$, so $u_j\to u$.
\end{proof}
