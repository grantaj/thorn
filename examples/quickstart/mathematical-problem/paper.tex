\begin{theorem}\label{thm:cancellation}
Let $a,b,c$ be real numbers. If $ac=bc$, then $a=b$.
\end{theorem}
\begin{proof}
From $ac=bc$, divide both sides by $c$. This gives $a=b$.
\end{proof}
